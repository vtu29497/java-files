public class W2_shuffle_array {

    public int[] shuffle(int[] nums, int n) {

        int[] ans = new int[nums.length];

        for (int i = 0; i < n; i++) {
            ans[2 * i] = nums[i];
            ans[2 * i + 1] = nums[n + i];
        }

        return ans;
    }

    public static void main(String[] args) {

        W2_shuffle_array obj = new W2_shuffle_array();

        int[] nums = {2, 5, 1, 3, 4, 7};
        int n = 3;

        int[] result = obj.shuffle(nums, n);

        for (int x : result) {
            System.out.print(x + " ");
        }
    }
}
